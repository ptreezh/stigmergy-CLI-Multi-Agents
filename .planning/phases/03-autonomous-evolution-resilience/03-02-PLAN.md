Create `src/core/soul/soul_scheduler.js` — plain CommonJS scheduler singleton enforcing single-instance evolution across processes (Criteria #5/#6). PID-file + TTL approach (cross-platform on Windows + POSIX):

```javascript
const fs = require('fs');
const path = require('path');

const DEFAULT_LOCK_PATH = path.join(
  process.env.HOME || process.env.USERPROFILE,
  '.stigmergy',
  'soul-state',
  'evolution.lock.json'
);
const DEFAULT_TTL_MS = 10 * 60 * 1000; // 10 minutes

class SoulScheduler {
  constructor(options = {}) {
    this.lockPath = options.lockPath || DEFAULT_LOCK_PATH;
    this.ttlMs = options.ttlMs || DEFAULT_TTL_MS;
  }

  /**
   * Try to acquire the evolution lock.
   * @returns {boolean} true when acquired (first-writer-wins); false when held by another live owner
   */
  acquire() {
    const dir = path.dirname(this.lockPath);
    fs.mkdirSync(dir, { recursive: true });

    if (fs.existsSync(this.lockPath)) {
      const lock = this._readLock();
      const expired = lock && (Date.now() - new Date(lock.startedAt).getTime()) > this.ttlMs;
      if (!expired) return false;          // held by a live owner
      this._stealExpiredLock();            // TTL auto-clear (Criterion #5)
    }

    try {
      // Exclusive create: fails if another process won the race (Criterion #6)
      const fd = fs.openSync(this.lockPath, 'wx');
      fs.writeFileSync(
        fd,
        JSON.stringify({ pid: process.pid, startedAt: new Date().toISOString() }, null, 2),
        'utf8'
      );
      fs.closeSync(fd);
      return true;
    } catch (err) {
      return false; // lost the first-writer race
    }
  }

  /** @returns {boolean} true when lock is currently held & not expired */
  isLocked() {
    if (!fs.existsSync(this.lockPath)) return false;
    const lock = this._readLock();
    if (!lock) return false;
    return (Date.now() - new Date(lock.startedAt).getTime()) <= this.ttlMs;
  }

  /** @returns {Object|null} { pid, startedAt } or null */
  getOwner() {
    return this._readLock();
  }

  release() {
    if (fs.existsSync(this.lockPath)) fs.unlinkSync(this.lockPath);
  }

  _readLock() {
    try {
      return JSON.parse(fs.readFileSync(this.lockPath, 'utf8'));
    } catch (err) {
      return null; // corrupt lock → treat as clearable
    }
  }

  _stealExpiredLock() {
    try { fs.unlinkSync(this.lockPath); } catch (err) { /* raced with another process */ }
  }
}

module.exports = { SoulScheduler, DEFAULT_LOCK_PATH, DEFAULT_TTL_MS };
```

Design constraints:
- CommonJS; NO third-party deps (no proper-lockfile/flock — pid+TTL is cross-platform, matches 03-RESEARCH.md assumption #3).
- `acquire()` is synchronous + atomic (openSync 'wx' = exclusive create). Callers await nothing.
- Corrupt lock file → treated as clearable (re-acquire next call) — threat T-3-02 mitigation.
- Instance is composable; a shared singleton export is NOT required at this stage — supervisor (wave 3) instantiates one.

After creation, fix `module.exports` shape ONLY if tests require `soulScheduler` named export — prefer named classes (`SoulScheduler`) matching codebase style (`DeadLetterQueue`, `CronScheduler`).
