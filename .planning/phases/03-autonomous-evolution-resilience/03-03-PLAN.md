Create `src/core/evolution/EvolutionSupervisor.js` — plain CommonJS class composing the wave-1/2 primitives. Coordinates the full resilient evolution loop:

```javascript
const path = require('path');
const { CircuitBreakerIntegration } = require('./CircuitBreakerIntegration');
const { CheckpointStore } = require('./CheckpointStore');
const { SoulScheduler } = require('../soul/soul_scheduler');
const { DeadLetterQueue } = require('../soul/DeadLetterQueue');
const { EventEmitter } = require('events');

const BACKOFF_BASE_MS = 1000;
const BACKOFF_CAP_MS = 60000;

class EvolutionSupervisor extends EventEmitter {
  constructor(options = {}) {
    super();
    this.skillEvolver = options.skillEvolver;          // required: SoulSkillEvolver instance
    this.breaker = options.breaker || new CircuitBreakerIntegration();
    this.checkpoints = options.checkpoints || new CheckpointStore();
    this.scheduler = options.scheduler || new SoulScheduler();
    this.dlq = options.dlq || new DeadLetterQueue();   // existing DLQ (adopt, not re-create)
    this.backoffBaseMs = options.backoffBaseMs || BACKOFF_BASE_MS;
    this.backoffCapMs = options.backoffCapMs || BACKOFF_CAP_MS;
    this.maxRetries = options.maxRetries || 15;        // ~ (retry budget across storm window)
    this._replayDLQOnStart();                          // Criterion #3/#4 wiring
  }

  /** Replay recoverable (ProcessError) entries from the DLQ before starting a fresh loop. */
  _replayDLQOnStart() {
    try {
      this.dlq.replay(async (entry) => {
        // Handler is invoked only for ProcessError entries (DLQ contract).
        // Re-dispatch via the same strategy the entry was recorded under.
        this.emit('dlq:retry', { id: entry.id, errorType: entry.errorType });
        await this.skillEvolver.recoverFromEntry?.(entry) ?? this._noop(entry);
      });
    } catch (err) {
      this.emit('dlq:replay:failed', { error: err.message });
    }
  }

  /**
   * Run one evolution direction with full resilience wrapping.
   * @param {string} direction — evolution direction passed to skillEvolver
   */
  async run(direction) {
    if (!this.scheduler.acquire()) {
      this.emit('scheduler:busy', { pid: this.scheduler.getOwner() });
      this.emit('alert', { severity: 'medium', message: 'evolution already running (lock held)' });
      return { status: 'skipped-lock' };
    }
    try {
      let attempt = 0;
      while (attempt <= this.maxRetries) {
        try {
          await this._executeStep(direction, attempt);
          this.breaker.recordSuccess(direction);
          return { status: 'ok', direction, attempts: attempt + 1 };
        } catch (err) {
          this.breaker.recordFailure(direction);
          this.dlq.push(err, { operation: 'evolve', direction });   // Criterion #3
          if (this.breaker.shouldSkipRetry(direction)) {
            this.emit('circuit:open', { direction });
            this.emit('alert', { severity: 'high', message: `circuit open for ${direction}` });
            return { status: 'circuit-open', direction, attempts: attempt + 1 };
          }
          if (attempt >= this.maxRetries) throw err;
          const delay = this._backoff(attempt);
          await new Promise((r) => setTimeout(r, delay));
          attempt++;
        }
      }
    } finally {
      this.scheduler.release();
    }
  }

  /** Run multiple directions sequentially under one lock. */
  async runBatch(directions) {
    const results = [];
    for (const dir of directions) results.push(await this.run(dir));
    return results;
  }

  /** Per-step execution with checkpoint. */
  async _executeStep(direction, attempt) {
    this.checkpoints.save({ lastCompletedStep: direction, attemptedStep: attempt, evolvedAt: new Date().toISOString() });
    await this.skillEvolver.evolve(direction);   // retryable operation (ProcessError flows from evolver)
    this.breaker.recordSuccess(direction);
    this.checkpoints.save({ lastCompletedStep: direction, evolvedAt: new Date().toISOString() });
    this.emit('step:complete', { direction });
  }

  _backoff(attempt) {
    return Math.min(this.backoffBaseMs * Math.pow(2, attempt), this.backoffCapMs);
  }
}

module.exports = { EvolutionSupervisor, BACKOFF_BASE_MS, BACKOFF_CAP_MS };
```

Design constraints:
- CommonJS, EventEmitter base (matches ErrorHandler + CronScheduler patterns); NO new dependencies.
- `this.dlq.replay()` — check actual signature at DeadLetterQueue.js:71 before invoking (handler receives entry; async handlers awaited by the DLQ OR handled internally — if replay does NOT await async handlers, wrap synchronization inside the handler and document it; do NOT change the DLQ).
- `recoverFromEntry` is OPTIONAL — if `SoulSkillEvolver` has no such method, replay handler only emits `dlq:retry` (retry-count bookkeeping still happens inside DLQ). Do not add dead methods to the evolver.
- Criterion #2: checkpoint saved BEFORE step (records what is being attempted) and AFTER step (records completion) — resume() in a real loop would restart from last COMPLETED step. Keep both saves; the after-save is the authoritative `lastCompletedStep`.
- Criterion #8: per-step try/catch inside loop; a single rejected step does NOT kill the loop; backoff doubles 1s→60s.
- Criterion #5/#6: acquire before, release in finally. `runBatch` loops directions under ONE lock.
- No `as any` / `@ts-ignore`; empty catch prohibited (the `try/catch` around replay is justified — logs via emit).
