Create `src/core/evolution/CircuitBreakerIntegration.js` — plain CommonJS class extending `EventEmitter` (no new dependencies). Per-strategy breaker state, in-memory:

```javascript
const { EventEmitter } = require('events');

const DEFAULT_FAILURE_THRESHOLD = 5;

class CircuitBreakerIntegration extends EventEmitter {
  constructor(options = {}) {
    super();
    this.failureThreshold = options.failureThreshold || DEFAULT_FAILURE_THRESHOLD;
    this._state = new Map(); // strategy -> { consecutiveFailures, open }
  }

  /**
   * @param {string} strategy — evolution strategy key (e.g. 'knowledge', 'skills', 'merge')
   * @returns {boolean} true when circuit is OPEN for this strategy (skip retries)
   */
  shouldSkipRetry(strategy) {
    const s = this._state.get(strategy);
    return !!(s && s.open);
  }

  /** @returns {boolean} true when circuit is OPEN */
  isOpen(strategy) {
    return this.shouldSkipRetry(strategy);
  }

  recordFailure(strategy) {
    const s = this._getStrategy(strategy);
    s.consecutiveFailures++;
    if (!s.open && s.consecutiveFailures >= this.failureThreshold) {
      s.open = true;
      this.emit('circuit:open', { strategy, consecutiveFailures: s.consecutiveFailures });
    }
    return s.open;
  }

  recordSuccess(strategy) {
    const s = this._getStrategy(strategy);
    const wasOpen = s.open;
    s.consecutiveFailures = 0;
    s.open = false;
    if (wasOpen) {
      this.emit('circuit:close', { strategy });
    }
  }

  reset(strategy) {
    this._state.delete(strategy);
  }

  resetAll() {
    this._state.clear();
  }

  _getStrategy(strategy) {
    let s = this._state.get(strategy);
    if (!s) {
      s = { consecutiveFailures: 0, open: false };
      this._state.set(strategy, s);
    }
    return s;
  }
}

module.exports = { CircuitBreakerIntegration, DEFAULT_FAILURE_THRESHOLD };
```

Design constraints:
- CommonJS (match codebase), EventEmitter base (matches ErrorHandler pattern for the wave-3 alert wiring).
- In-memory only (single-process trust boundary per threat model); NO file I/O in this module.
- No `as any` / `@ts-ignore`; contextual error messages only.

Create `src/core/evolution/CheckpointStore.js` — plain CommonJS class. Idempotent checkpoint persistence with atomic writes:

```javascript
const fs = require('fs');
const path = require('path');

const DEFAULT_CHECKPOINT_PATH = path.join(
  process.env.HOME || process.env.USERPROFILE,
  '.stigmergy',
  'soul-state',
  'evolution-checkpoint.json'
);

class CheckpointStore {
  constructor(options = {}) {
    this.checkpointPath = options.checkpointPath || DEFAULT_CHECKPOINT_PATH;
  }

  /**
   * Persist checkpoint atomically (tmp + renameSync).
   * @param {Object} checkpoint — { lastCompletedStep, evolutionHistory, evolvedAt }
   */
  save(checkpoint) {
    const dir = path.dirname(this.checkpointPath);
    fs.mkdirSync(dir, { recursive: true });
    const tmpPath = this.checkpointPath + '.tmp';
    fs.writeFileSync(tmpPath, JSON.stringify(checkpoint, null, 2), 'utf8');
    fs.renameSync(tmpPath, this.checkpointPath); // atomic on same volume
  }

  /**
   * @returns {Object|null} checkpoint, or null when missing/corrupt (fresh start)
   */
  resume() {
    if (!fs.existsSync(this.checkpointPath)) return null;
    try {
      const parsed = JSON.parse(fs.readFileSync(this.checkpointPath, 'utf8'));
      return parsed && typeof parsed === 'object' ? parsed : null;
    } catch (err) {
      return null; // corrupt → fresh start (threat T-3-01/02 mitigation)
    }
  }

  /** @returns {string|null} last completed step id, or null */
  getLastStep() {
    const c = this.resume();
    return c ? c.lastCompletedStep : null;
  }

  clear() {
    if (fs.existsSync(this.checkpointPath)) fs.unlinkSync(this.checkpointPath);
  }
}

module.exports = { CheckpointStore, DEFAULT_CHECKPOINT_PATH };
```

Design constraints:
- CommonJS; tmp+renameSync atomic pattern EXACTLY matches existing codebase (`soul_skill_evolver.js:641-643`, `DeadLetterQueue.js:101`) — same-volume rename is atomic on Windows NTFS and POSIX.
- Corrupt JSON → `null` (fresh start), never throws (threat T-3-01).
- Constructor accepts custom path → tests use OS temp dir (isolation, no user-state pollution).
- No `as any` / `@ts-ignore`; no empty catch — corrupt-parse branch returns null deliberately.
